Go-IM 即时通讯系统 - 开发文档
================================

项目概述
---------

基于 Go + WebSocket 的即时通讯系统，支持用户认证、好友管理、群组聊天、私聊。

技术栈
-------

Web框架: github.com/gin-gonic/gin
WebSocket: github.com/gorilla/websocket
ORM: gorm.io/gorm + gorm.io/driver/mysql
缓存: github.com/go-redis/redis/v9
JWT: github.com/golang-jwt/jwt/v5
配置: github.com/spf13/viper
密码加密: golang.org/x/crypto/bcrypt
日志: github.com/sirupsen/logrus
UUID: github.com/google/uuid

二、MySQL 数据模型
------------------

-- 1. 用户表
CREATE TABLE `users` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(50) NOT NULL UNIQUE,
`password` varchar(255) NOT NULL,
`nickname` varchar(50) NOT NULL DEFAULT '',
`avatar` varchar(255) DEFAULT '',
`status` tinyint(1) DEFAULT '0' COMMENT '0-离线,1-在线',
`last_login` datetime DEFAULT NULL,
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
`updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
`deleted_at` datetime DEFAULT NULL,
PRIMARY KEY (`id`),
KEY `idx_username` (`username`)
);

-- 2. 好友关系表
CREATE TABLE `friends` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`user_id` int(11) NOT NULL,
`friend_id` int(11) NOT NULL,
`status` tinyint(1) DEFAULT '0' COMMENT '0-待确认,1-已确认',
`remark` varchar(50) DEFAULT '',
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
`updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
PRIMARY KEY (`id`),
KEY `idx_user_id` (`user_id`),
KEY `idx_friend_id` (`friend_id`)
);

-- 3. 群组表
CREATE TABLE `groups` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(50) NOT NULL,
`avatar` varchar(255) DEFAULT '',
`owner_id` int(11) NOT NULL,
`notice` varchar(255) DEFAULT '',
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
`updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
PRIMARY KEY (`id`),
KEY `idx_owner_id` (`owner_id`)
);

-- 4. 群成员表
CREATE TABLE `group_members` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`group_id` int(11) NOT NULL,
`user_id` int(11) NOT NULL,
`role` tinyint(1) DEFAULT '0' COMMENT '0-成员,1-管理员,2-群主',
`join_time` datetime DEFAULT NULL,
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
PRIMARY KEY (`id`),
KEY `idx_group_id` (`group_id`),
KEY `idx_user_id` (`user_id`)
);

-- 5. 消息表
CREATE TABLE `messages` (
`id` bigint(20) NOT NULL AUTO_INCREMENT,
`msg_id` varchar(36) NOT NULL,
`sender_id` int(11) NOT NULL,
`receiver_id` int(11) NOT NULL,
`msg_type` tinyint(1) DEFAULT '1' COMMENT '1-文本,2-图片,3-语音',
`chat_type` tinyint(1) DEFAULT '1' COMMENT '1-私聊,2-群聊',
`content` text,
`status` tinyint(1) DEFAULT '0' COMMENT '0-未读,1-已读',
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
PRIMARY KEY (`id`),
KEY `idx_msg_id` (`msg_id`),
KEY `idx_sender_id` (`sender_id`),
KEY `idx_receiver_id` (`receiver_id`),
KEY `idx_created_at` (`created_at`)
);

-- 6. 好友请求表
CREATE TABLE `friend_requests` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`from_user_id` int(11) NOT NULL,
`to_user_id` int(11) NOT NULL,
`message` varchar(255) DEFAULT '',
`status` tinyint(1) DEFAULT '0' COMMENT '0-待处理,1-已同意,2-已拒绝',
`created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
`updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
PRIMARY KEY (`id`),
KEY `idx_from_user_id` (`from_user_id`),
KEY `idx_to_user_id` (`to_user_id`)
);

三、Redis 数据结构
-------------------

Key: user:online:{user_id} 类型: String 说明: 在线状态 1=在线
Key: user:token:{user_id} 类型: String 说明: JWT Token, TTL: 24h
Key: user:socket:{user_id} 类型: String 说明: WebSocket连接ID
Key: msg:unread:{user_id} 类型: Set 说明: 未读消息ID列表
Key: chat:history:{chat_id} 类型: List 说明: 最近消息, TTL: 7天
Key: rate:limit:{ip} 类型: String 说明: IP限流, TTL: 1min

四、API 接口清单
-----------------

模块1: 用户认证 (Auth)
-----------------------

POST /api/v1/auth/register 注册 Body: {username, password, nickname}
POST /api/v1/auth/login 登录 Body: {username, password}
POST /api/v1/auth/logout 登出 Header: Authorization
GET /api/v1/auth/me 获取用户 Header: Authorization

模块2: 好友管理 (Friend)
-------------------------

GET /api/v1/friends 获取好友列表
POST /api/v1/friends/requests 发送好友申请
GET /api/v1/friends/requests 获取好友申请列表
PUT /api/v1/friends/requests/:id 处理好友申请
DELETE /api/v1/friends/:id 删除好友
PUT /api/v1/friends/:id/remark 修改好友备注

模块3: 群组管理 (Group)
-------------------------

POST /api/v1/groups 创建群组
GET /api/v1/groups 获取我的群组列表
GET /api/v1/groups/:id 获取群组详情
POST /api/v1/groups/:id/join 加入群组
POST /api/v1/groups/:id/leave 退出群组
PUT /api/v1/groups/:id 修改群信息

模块4: 消息管理 (Message)
---------------------------

GET /api/v1/messages/private/:friend_id 获取私聊历史
GET /api/v1/messages/group/:group_id 获取群聊历史
PUT /api/v1/messages/read/:msg_id 标记消息已读

五、WebSocket 消息协议
----------------------

连接地址: ws://localhost:8080/ws?token={jwt_token}

消息格式:
{
"type": "private_message", // 消息类型
"data": {
"msg_id": "uuid",
"sender_id": 1,
"receiver_id": 2,
"content": "hello",
"msg_type": 1, // 1-文本, 2-图片, 3-语音
"timestamp": 1234567890
}
}

消息类型列表:
private_message - 私聊消息
group_message - 群聊消息
typing - 正在输入
read_receipt - 已读回执
online_status - 在线状态变更(服务端推送)
friend_request - 好友申请通知(服务端推送)
message_ack - 消息确认

package service

type AuthService interface {
Register(username, password, nickname string) (*model.User, error)
Login(username, password string) (string,*model.User, error)
Logout(userID uint) error
GetUserByID(userID uint) (*model.User, error)
}

type FriendService interface {
ListFriends(userID uint) ([]*FriendInfo, error)
SendRequest(fromUserID, toUserID uint, message string) error
HandleRequest(requestID uint, status string) error
DeleteFriend(userID, friendID uint) error
GetRequests(userID uint) ([]*FriendRequest, error)
}

type GroupService interface {
CreateGroup(ownerID uint, name, avatar, notice string) (*model.Group, error)
JoinGroup(userID, groupID uint) error
LeaveGroup(userID, groupID uint) error
GetGroups(userID uint) ([]*GroupInfo, error)
GetGroupDetail(groupID uint) (*GroupDetail, error)
}

type MessageService interface {
SendPrivateMessage(senderID, receiverID uint, content string, msgType int) (*model.Message, error)
SendGroupMessage(senderID, groupID uint, content string, msgType int) (*model.Message, error)
GetPrivateHistory(userID, friendID uint, page, size int) ([]*model.Message, int64, error)
GetGroupHistory(userID, groupID uint, page, size int) ([]*model.Message, int64, error)
MarkAsRead(userID, msgID uint) error
GetUnreadMessages(userID uint) ([]*model.Message, error)
}

十、前端 WebSocket 对接示例
---------------------------

// 连接
const ws = new WebSocket('ws://localhost:8080/ws?token=' + token);

// 发送消息
ws.send(JSON.stringify({
type: 'private_message',
data: { receiver_id: 2, content: 'hello', msg_type: 1 }
}));

// 接收消息
ws.onmessage = (event) => {
const msg = JSON.parse(event.data);
console.log(msg);
};

// HTTP请求携带Token
fetch('/api/v1/auth/me', {
headers: { 'Authorization': 'Bearer ' + token }
});
