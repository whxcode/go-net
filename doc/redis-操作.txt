========================================
go-redis 常用操作速查
========================================

import "github.com/redis/go-redis/v9"


一、String（字符串）
========================================
// 存
err := rdb.Set(ctx, "key", "value", 0).Err()
err := rdb.Set(ctx, "key", "value", 7*24*time.Hour).Err()  // 带过期时间

// 取
val, err := rdb.Get(ctx, "key").Result()
val, err := rdb.Get(ctx, "key").Int64()      // 转 int64
val, err := rdb.Get(ctx, "key").Uint64()     // 转 uint64
val, err := rdb.Get(ctx, "key").Bool()       // 转 bool

// 删
err := rdb.Del(ctx, "key").Err()

// 判断是否存在
val, err := rdb.Exists(ctx, "key").Result()  // 返回 1 存在，0 不存在

// 自增/自减
val, err := rdb.Incr(ctx, "counter").Result()
val, err := rdb.IncrBy(ctx, "counter", 10).Result()
val, err := rdb.Decr(ctx, "counter").Result()

// 设置过期时间
err := rdb.Expire(ctx, "key", 7*24*time.Hour).Err()

// 查看过期时间
ttl, err := rdb.TTL(ctx, "key").Result()  // 返回 time.Duration


二、Hash（哈希表）
========================================
// 存字段
err := rdb.HSet(ctx, "user:1", "name", "张三").Err()
err := rdb.HSet(ctx, "user:1", "name", "张三", "age", 18).Err()

// 取字段
val, err := rdb.HGet(ctx, "user:1", "name").Result()

// 取全部字段
vals, err := rdb.HGetAll(ctx, "user:1").Result()  // 返回 map[string]string

// 删除字段
err := rdb.HDel(ctx, "user:1", "age").Err()

// 判断字段是否存在
ok, err := rdb.HExists(ctx, "user:1", "name").Result()  // 返回 bool

// 字段自增
val, err := rdb.HIncrBy(ctx, "user:1", "age", 1).Result()


三、List（列表）
========================================
// 从左边插入
err := rdb.LPush(ctx, "queue", "msg1").Err()
err := rdb.LPush(ctx, "queue", "msg2", "msg3").Err()

// 从右边插入
err := rdb.RPush(ctx, "queue", "msg1").Err()

// 从左边弹出
val, err := rdb.LPop(ctx, "queue").Result()

// 从右边弹出
val, err := rdb.RPop(ctx, "queue").Result()

// 取指定范围（0 -1 取全部）
vals, err := rdb.LRange(ctx, "queue", 0, -1).Result()
vals, err := rdb.LRange(ctx, "queue", 0, 9).Result()  // 取前10条

// 列表长度
n, err := rdb.LLen(ctx, "queue").Result()


四、Set（集合）
========================================
// 添加元素
err := rdb.SAdd(ctx, "tags", "go", "redis", "gin").Err()

// 获取所有元素
vals, err := rdb.SMembers(ctx, "tags").Result()

// 判断是否存在
ok, err := rdb.SIsMember(ctx, "tags", "go").Result()

// 删除元素
err := rdb.SRem(ctx, "tags", "go").Err()

// 集合大小
n, err := rdb.SCard(ctx, "tags").Result()

// 交集
vals, err := rdb.SInter(ctx, "set1", "set2").Result()

// 并集
vals, err := rdb.SUnion(ctx, "set1", "set2").Result()


五、Sorted Set（有序集合）
========================================
// 添加元素
err := rdb.ZAdd(ctx, "rank", redis.Z{Score: 100, Member: "张三"}).Err()
err := rdb.ZAdd(ctx, "rank", 
    redis.Z{Score: 100, Member: "张三"},
    redis.Z{Score: 90, Member: "李四"},
).Err()

// 按分数升序取（从低到高）
vals, err := rdb.ZRange(ctx, "rank", 0, -1).Result()
// 按分数降序取（从高到低）
vals, err := rdb.ZRevRange(ctx, "rank", 0, -1).Result()

// 取前10名（从高到低）
vals, err := rdb.ZRevRange(ctx, "rank", 0, 9).Result()

// 取分数
score, err := rdb.ZScore(ctx, "rank", "张三").Result()

// 增加分数
val, err := rdb.ZIncrBy(ctx, "rank", 10, "张三").Result()

// 删除
err := rdb.ZRem(ctx, "rank", "张三").Err()

// 排名（从0开始）
n, err := rdb.ZRank(ctx, "rank", "张三").Result()       // 正序排名
n, err := rdb.ZRevRank(ctx, "rank", "张三").Result()    // 倒序排名


六、Pub/Sub（发布订阅）
========================================
// 发布
err := rdb.Publish(ctx, "channel", "hello").Err()

// 订阅
pubsub := rdb.Subscribe(ctx, "channel")
defer pubsub.Close()

// 接收消息
for {
    msg, err := pubsub.ReceiveMessage(ctx)
    if err != nil {
        break
    }
    fmt.Println(msg.Payload)
}

// 带超时接收
msg, err := pubsub.ReceiveMessage(ctx)
if err == redis.Nil {
    // 超时
}


七、Pipeline（批量操作）
========================================
pipe := rdb.Pipeline()
pipe.Set(ctx, "key1", "val1", 0)
pipe.Set(ctx, "key2", "val2", 0)
pipe.Incr(ctx, "counter")
cmds, err := pipe.Exec(ctx)


八、通用
========================================
// 删除 key
err := rdb.Del(ctx, "key1", "key2").Err()

// 判断 key 是否存在
n, err := rdb.Exists(ctx, "key").Result()

// 设置过期时间
err := rdb.Expire(ctx, "key", 7*24*time.Hour).Err()

// 查看过期时间
ttl, err := rdb.TTL(ctx, "key").Result()

// 查看 key 类型
typ, err := rdb.Type(ctx, "key").Result()

十、完整示例
========================================
package model

import (
    "context"
    "time"
    "github.com/redis/go-redis/v9"
)

var Ctx = context.Background()
var Rdb *redis.Client

func InitRedis() {
    Rdb = redis.NewClient(&redis.Options{
        Addr:     "localhost:6379",
        Password: "",
        DB:       0,
    })
}

// token 存储
func SetToken(token string, userID uint) error {
    return Rdb.Set(Ctx, token, userID, 7*24*time.Hour).Err()
}

func GetToken(token string) (string, error) {
    return Rdb.Get(Ctx, token).Result()
}

func DelToken(token string) error {
    return Rdb.Del(Ctx, token).Err()
}


========================================
Redis 原始命令速查（常用运维）
========================================

一、Key 操作（通用）
========================================
KEYS *                    # 查看所有 key（生产慎用）
KEYS user:*               # 查看匹配的 key
EXISTS key                # 判断 key 是否存在
TYPE key                  # 查看 key 的类型
TTL key                   # 查看剩余过期时间（秒）
PTTL key                  # 查看剩余过期时间（毫秒）
EXPIRE key 3600           # 设置过期时间（秒）
PEXPIRE key 3600000       # 设置过期时间（毫秒）
PERSIST key               # 移除过期时间
RENAME old new            # 重命名 key
DEL key1 key2             # 删除 key（可多个）


二、清空数据库
========================================
FLUSHDB                   # 清空当前数据库
FLUSHALL                  # 清空所有数据库
DBSIZE                    # 查看当前数据库 key 数量


三、String
========================================
SET key value             # 存
GET key                   # 取
SETEX key 3600 value      # 存并设置过期时间（秒）
SETNX key value           # 不存在才存
INCR key                  # 自增 1
INCRBY key 10             # 自增指定值
DECR key                  # 自减 1


四、Hash
========================================
HSET key field value      # 存字段
HGET key field            # 取字段
HGETALL key               # 取所有字段
HDEL key field            # 删除字段
HEXISTS key field         # 判断字段是否存在
HLEN key                  # 字段数量


五、List
========================================
LPUSH key value           # 左边插入
RPUSH key value           # 右边插入
LPOP key                  # 左边弹出
RPOP key                  # 右边弹出
LRANGE key 0 -1           # 取所有
LLEN key                  # 列表长度


六、Set
========================================
SADD key member           # 添加
SMEMBERS key              # 取所有
SISMEMBER key member      # 判断是否存在
SREM key member           # 删除
SCARD key                 # 集合大小


七、ZSet
========================================
ZADD key score member     # 添加
ZRANGE key 0 -1           # 升序取所有
ZREVRANGE key 0 -1        # 降序取所有
ZREM key member           # 删除


八、查看服务器信息

九、调试（生产慎用）

========================================
Redis 数据库切换命令速查
========================================

一、切换数据库
========================================
SELECT 0        # 切换到 0 号数据库
SELECT 1        # 切换到 1 号数据库
SELECT 15       # 切换到 15 号数据库（最大 15）


二、查看当前数据库
========================================
SELECT 0        # 切换后不会显示当前是几号库
CLIENT INFO     # 查看当前客户端信息（包含 db 编号）

# 或者在 redis-cli 连接时显示
redis-cli -n 1  # 直接连接 1 号数据库


三、查看数据库信息
========================================
DBSIZE          # 当前数据库 key 数量
INFO keyspace   # 查看所有数据库的统计信息

# 输出示例：
# db0:keys=100,expires=10,avg_ttl=3600
# db1:keys=50,expires=5,avg_ttl=7200


四、清空数据库
========================================
FLUSHDB         # 清空当前数据库
FLUSHALL        # 清空所有数据库（慎用！）


五、连接时指定数据库
========================================
# redis-cli 方式
redis-cli -n 1                      # 连接 1 号数据库
redis-cli -h 127.0.0.1 -p 6379 -n 1 # 指定主机端口和数据库

# Go 代码方式
rdb := redis.NewClient(&redis.Options{
    Addr: "localhost:6379",
    DB:   1,  // 指定 1 号数据库
})


六、查看当前连接在哪个数据库
========================================
# redis-cli 中
CLIENT INFO | grep db

# 或者直接
INFO clients
